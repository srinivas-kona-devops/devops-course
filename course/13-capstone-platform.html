3. Step 1: Terraform builds EKS

bash
cd ~/lab/platform/infra
cat > main.tf <<'EOT'
terraform {
  required_version = ">= 1.6"
  required_providers {
    aws = { source = "hashicorp/aws", version = "~> 5.0" }
  }
}
provider "aws" { region = var.region }

variable "region" {
  type    = string
  default = "ap-south-1"
}
variable "cluster_name" {
  type    = string
  default = "platform-lab"
}

data "aws_availability_zones" "az" { state = "available" }

module "vpc" {
  source  = "terraform-aws-modules/vpc/aws"
  version = "~> 5.0"

  name               = var.cluster_name
  cidr               = "10.0.0.0/16"
  azs                = slice(data.aws_availability_zones.az.names, 0, 2)
  public_subnets     = ["10.0.1.0/24", "10.0.2.0/24"]
  private_subnets    = ["10.0.11.0/24", "10.0.12.0/24"]
  enable_nat_gateway = true
  single_nat_gateway = true # cheaper for a lab; use one per AZ in production
  public_subnet_tags  = { "kubernetes.io/role/elb" = 1 }
  private_subnet_tags = { "kubernetes.io/role/internal-elb" = 1 }
}

module "eks" {
  source  = "terraform-aws-modules/eks/aws"
  version = "~> 20.0"

  cluster_name    = var.cluster_name
  cluster_version = "1.31" # choose a version currently supported by EKS
  vpc_id          = module.vpc.vpc_id
  subnet_ids      = module.vpc.private_subnets

  cluster_endpoint_public_access           = true
  enable_cluster_creator_admin_permissions = true

  eks_managed_node_groups = {
    default = {
      instance_types = ["t3.large"]
      min_size       = 2
      max_size       = 3
      desired_size   = 2
    }
  }
}

resource "aws_ecr_repository" "app" {
  name                 = "todo-api"
  image_tag_mutability = "IMMUTABLE"
  force_delete         = true # lab only
  image_scanning_configuration { scan_on_push = true }
}

output "cluster_name" { value = module.eks.cluster_name }
output "ecr_url"      { value = aws_ecr_repository.app.repository_url }
EOT
terraform init
terraform validate
terraform plan -out tfplan
terraform apply tfplan                # about 15-20 minutes
aws eks update-kubeconfig --name $(terraform output -raw cluster_name) --region ap-south-1
kubectl get nodes
Why IMMUTABLE tags
An immutable tag can never be overwritten, so a tag always identifies one exact image. Combined with a build-number tag, this makes rollbacks trustworthy.

bash

EOT

- name: Prepare CI server
  hosts: ci
  become: true
  vars: {k8s_minor: "v1.31"}
  tasks:
  - name: Base packages
    ansible.builtin.apt: {name: [docker.io, git, curl, unzip, jq, openjdk-17-jre-headless], state: present, update_cache: true}
  - name: Docker running
    ansible.builtin.service: {name: docker, state: started, enabled: true}
  - name: kubectl and helm
    ansible.builtin.shell: |
      curl -fsSL -o /usr/local/bin/kubectl "https://dl.k8s.io/release/$(curl -fsSL https://dl.k8s.io/release/stable.txt)/bin/linux/amd64/kubectl" && chmod +x /usr/local/bin/kubectl
      curl -fsSL https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3 | bash
    args: {creates: /usr/local/bin/helm}
  - name: Jenkins container
    community.docker.docker_container:
      name: jenkins
      image: jenkins/jenkins:lts-jdk17
      restart_policy: unless-stopped
      published_ports: ["8088:8080"]
      volumes: [jenkins_home:/var/jenkins_home]
EOT

6. Step 4: Helm chart with canary support

Add a second Deployment (the canary) and let Istio split traffic. In app/chart/templates/istio.yaml:

bash
cd ~/lab/platform/app
cat >> chart/values.yaml <<'EOT'
canary: {enabled: false, tag: "", weight: 10}
EOT
cat > chart/templates/istio.yaml <<'EOT'
apiVersion: networking.istio.io/v1
kind: DestinationRule
metadata: {name: {{ .Release.Name }}-api}
spec:
  host: {{ .Release.Name }}-api
  subsets:
  - {name: stable, labels: {track: stable}}
  - {name: canary, labels: {track: canary}}
---
apiVersion: networking.istio.io/v1
kind: VirtualService
metadata: {name: {{ .Release.Name }}-api}
spec:
  hosts: [{{ .Release.Name }}-api]
  http:
  - route:
    - {destination: {host: {{ .Release.Name }}-api, subset: stable}, weight: {{ sub 100 (.Values.canary.enabled | ternary (int .Values.canary.weight) 0) }}}
{{- if .Values.canary.enabled }}
    - {destination: {host: {{ .Release.Name }}-api, subset: canary}, weight: {{ .Values.canary.weight }}}
{{- end }}
EOT
# Edit chart/templates/api.yaml: add label "track: stable" to pod template labels.
# Add a copy of the api Deployment named {{ .Release.Name }}-api-canary with track: canary,
# image tag .Values.canary.tag, wrapped in {{- if .Values.canary.enabled }} ... {{- end }}.
# The Service selects only app=..., so both tracks receive traffic and Istio weights decide the split.
helm lint chart --set db.password=x
Where the traffic is decided
Kubernetes Services spread evenly by pod count; Istio's VirtualService overrides that with exact percentages, so one canary pod can receive 10% regardless of how many stable pods exist.

7. Step 5: The Jenkinsfile

groovy
pipeline {
  agent any
  environment {
    AWS_REGION = 'ap-south-1'
    ECR        = credentials('ecr-registry-url')   // text credential: ACCOUNT.dkr.ecr.REGION.amazonaws.com
    TAG        = "${env.GIT_COMMIT.take(8)}-${env.BUILD_NUMBER}"
  }
  stages {
    stage('Test')  { steps { sh 'cd app/api && python3 -m py_compile app.py' } }
    stage('Build') { steps { sh 'docker build -t $ECR/todo-api:$TAG app/api' } }
    stage('Scan')  { steps { sh 'trivy image --exit-code 1 --severity CRITICAL $ECR/todo-api:$TAG || true' } }
    stage('Push')  {
      steps {
        sh 'aws ecr get-login-password --region $AWS_REGION | docker login --username AWS --password-stdin $ECR'
        sh 'docker push $ECR/todo-api:$TAG'
      }
    }
    stage('Canary 10%') {
      steps {
        sh '''aws eks update-kubeconfig --name platform-lab --region $AWS_REGION
              helm upgrade --install shop app/chart -n shop --reuse-values \
                --set api.image.repository=$ECR/todo-api --set canary.enabled=true \
                --set canary.tag=$TAG --set canary.weight=10 --wait --timeout 5m'''
      }
    }
    stage('Verify canary') {
      steps {
        sh '''sleep 60
              ERR=$(kubectl -n istio-system exec deploy/prometheus -- wget -qO- \
                "http://localhost:9090/api/v1/query?query=sum(rate(istio_requests_total{destination_workload=~\\".*canary\\",response_code=~\\"5..\\"}[1m]))" | jq -r '.data.result[0].value[1] // "0"')
              echo "canary 5xx rate: $ERR"; [ "$(echo "$ERR < 0.05" | bc)" = 1 ]'''
      }
    }
    stage('Promote') {
      steps {
        input message: 'Promote canary to 100%?'
        sh 'helm upgrade shop app/chart -n shop --reuse-values --set api.image.tag=$TAG --set canary.enabled=false --wait'
      }
    }
  }
  post { failure { sh 'helm rollback shop -n shop || true' } }
}
Gate on evidence
The pipeline promotes only if the canary's 5xx rate stays under the threshold and a human approves. Any failure triggers helm rollback.

9. Clean up (mandatory)

bash
helm uninstall shop -n shop; kubectl delete ns shop
istioctl uninstall --purge -y; kubectl delete ns istio-system    # removes the AWS load balancer first
sleep 60
cd ~/lab/platform/infra && terraform destroy
aws elbv2 describe-load-balancers --query 'LoadBalancers[].LoadBalancerName'   # should be empty
aws ec2 describe-nat-gateways --filter Name=state,Values=available --query 'NatGateways[].NatGatewayId'
Order matters
Delete Kubernetes load balancers before terraform destroy. Otherwise AWS keeps ENIs in the VPC and the destroy hangs on subnet deletion.
Checkpoint
You have provisioned a cluster from code, configured a server with Ansible, shipped an image through a pipeline, shifted 10% of traffic with Istio, and torn everything down cleanly.
