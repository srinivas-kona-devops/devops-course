2. Install Docker Engine (official repository)

bash
install -m 0755 -d /etc/apt/keyrings
curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.asc] https://download.docker.com/linux/ubuntu $(. /etc/os-release && echo $VERSION_CODENAME) stable" > /etc/apt/sources.list.d/docker.list
apt update && apt -y install docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin
systemctl enable --now docker
usermod -aG docker devops
docker version; docker run --rm hello-world

4. Build an image

bash
mkdir -p ~/lab/app && cd ~/lab/app
cat > app.py <<'EOT'
from http.server import BaseHTTPRequestHandler, HTTPServer
class H(BaseHTTPRequestHandler):
    def do_GET(self):
        self.send_response(200); self.end_headers(); self.wfile.write(b"hello devops\n")
HTTPServer(("", 8000), H).serve_forever()
EOT
cat > Dockerfile <<'EOT'
FROM python:3.12-slim
RUN useradd -m app
WORKDIR /app
COPY app.py .
USER app
EXPOSE 8000
CMD ["python", "app.py"]
EOT
docker build -t hello:1.0 .
docker run -d --name hello -p 8000:8000 hello:1.0
curl localhost:8000; docker history hello:1.0; docker rm -f hello
Best practices
Pin base image versions, run as non-root, add a .dockerignore, order layers from least to most frequently changed, and use multi-stage builds to keep images small.

5. Multi-stage build

dockerfile
FROM golang:1.23 AS build
WORKDIR /src
COPY . .
RUN CGO_ENABLED=0 go build -o /out/app .
FROM gcr.io/distroless/static
COPY --from=build /out/app /app
ENTRYPOINT ["/app"]

6. Volumes and networks

bash
docker volume create data
docker run --rm -v data:/d alpine sh -c 'echo persisted > /d/f'
docker run --rm -v data:/d alpine cat /d/f
docker network create appnet
docker run -d --name db --network appnet -e POSTGRES_PASSWORD=pw postgres:16-alpine
docker run --rm --network appnet alpine nslookup db
docker rm -f db; docker network rm appnet; docker volume rm data

7. Docker Compose

bash
cd ~/lab/app
cat > compose.yaml <<'EOT'
services:
  web:
    build: .
    ports: ["8000:8000"]
    depends_on: [cache]
  cache:
    image: redis:7-alpine
EOT
docker compose up -d --build
docker compose ps; curl localhost:8000
docker compose logs --tail 3; docker compose down

8. Local registry, cleanup, hardening

bash
docker run -d -p 5000:5000 --name registry registry:2
docker tag hello:1.0 localhost:5000/hello:1.0 && docker push localhost:5000/hello:1.0
curl -s localhost:5000/v2/_catalog; docker rm -f registry
docker system df; docker system prune -f
docker run --rm --read-only --cap-drop ALL --security-opt no-new-privileges alpine id
Security
Membership of the docker group is equivalent to root. Never mount /var/run/docker.sock into untrusted containers.
