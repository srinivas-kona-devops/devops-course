2. Source code and Git

bash
mkdir -p ~/lab/todo/api && cd ~/lab/todo && git init -q
cat > api/app.py <<'EOT'
import os, psycopg2
from flask import Flask, jsonify, request
app = Flask(__name__)
def db():
    return psycopg2.connect(host=os.environ["DB_HOST"], user="postgres",
                            password=os.environ["DB_PASSWORD"], dbname="postgres", connect_timeout=3)
def init():
    with db() as c, c.cursor() as cur:
        cur.execute("create table if not exists todos(id serial primary key, title text not null)")
@app.get("/healthz")
def healthz(): return "ok"
@app.get("/readyz")
def readyz():
    try:
        with db() as c, c.cursor() as cur: cur.execute("select 1")
        return "ready"
    except Exception as e:
        return str(e), 503
@app.get("/api/todos")
def list_todos():
    with db() as c, c.cursor() as cur:
        cur.execute("select id, title from todos order by id")
        return jsonify([{"id": i, "title": t} for i, t in cur.fetchall()])
@app.post("/api/todos")
def add_todo():
    with db() as c, c.cursor() as cur:
        cur.execute("insert into todos(title) values (%s) returning id", (request.json["title"],))
        return jsonify({"id": cur.fetchone()[0]}), 201
if __name__ == "__main__":
    init()
    app.run(host="0.0.0.0", port=8000)
EOT
printf 'flask==3.0.3\npsycopg2-binary==2.9.9\ngunicorn==23.0.0\n' > api/requirements.txt
cat > api/Dockerfile <<'EOT'
FROM python:3.12-slim
ENV PYTHONUNBUFFERED=1
RUN useradd -m -u 10001 app
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY app.py .
USER 10001
EXPOSE 8000
CMD ["python", "app.py"]
EOT
printf '__pycache__/\n*.pyc\n.env\n' > .gitignore
git add -A && git commit -qm "feat: todo api" && git log --oneline

3. Build and test with Compose first

bash
cd ~/lab/todo
cat > compose.yaml <<'EOT'
services:
  db:
    image: postgres:16-alpine
    environment: {POSTGRES_PASSWORD: devpass}
    healthcheck: {test: ["CMD-SHELL", "pg_isready -U postgres"], interval: 3s, retries: 10}
  api:
    build: ./api
    environment: {DB_HOST: db, DB_PASSWORD: devpass}
    ports: ["8000:8000"]
    depends_on: {db: {condition: service_healthy}}
EOT
docker compose up -d --build
sleep 5
curl -s -XPOST localhost:8000/api/todos -H 'content-type: application/json' -d '{"title":"learn devops"}'
curl -s localhost:8000/api/todos; echo
docker compose down -v

5. Helm chart for the whole app

bash
cd ~/lab/todo && helm create chart && rm -rf chart/templates/* chart/charts
cat > chart/Chart.yaml <<'EOT'
apiVersion: v2
name: todo
description: Three-tier to-do application
type: application
version: 0.1.0
appVersion: "1.0.0"
EOT
cat > chart/values.yaml <<'EOT'
api:
  image: {repository: todo-api, tag: "1.0.0", pullPolicy: IfNotPresent}
  replicas: 3
  resources: {requests: {cpu: 50m, memory: 96Mi}, limits: {cpu: 300m, memory: 192Mi}}
db:
  image: postgres:16-alpine
  storage: 1Gi
  password: ""        # set with --set db.password=... or an existing secret
  existingSecret: ""
web:
  image: nginxinc/nginx-unprivileged:1.27-alpine
ingress: {enabled: true, className: nginx, host: todo.lab.local}
autoscaling: {enabled: true, min: 2, max: 6, cpu: 60}
EOT
cat > chart/templates/_helpers.tpl <<'EOT'
{{- define "todo.secretName" -}}
{{- default (printf "%s-db" .Release.Name) .Values.db.existingSecret -}}
{{- end -}}
EOT
cat > chart/templates/db.yaml <<'EOT'
{{- if not .Values.db.existingSecret }}
apiVersion: v1
kind: Secret
metadata: {name: {{ .Release.Name }}-db}
stringData: {password: {{ required "set db.password" .Values.db.password | quote }}}
---
{{- end }}
apiVersion: v1
kind: Service
metadata: {name: {{ .Release.Name }}-db}
spec: {clusterIP: None, selector: {app: {{ .Release.Name }}-db}, ports: [{port: 5432}]}
---
apiVersion: apps/v1
kind: StatefulSet
metadata: {name: {{ .Release.Name }}-db}
spec:
  serviceName: {{ .Release.Name }}-db
  replicas: 1
  selector: {matchLabels: {app: {{ .Release.Name }}-db}}
  template:
    metadata: {labels: {app: {{ .Release.Name }}-db}}
    spec:
      containers:
      - name: pg
        image: {{ .Values.db.image }}
        env:
        - {name: POSTGRES_PASSWORD, valueFrom: {secretKeyRef: {name: {{ include "todo.secretName" . }}, key: password}}}
        - {name: PGDATA, value: /var/lib/postgresql/data/pgdata}
        readinessProbe: {exec: {command: ["pg_isready","-U","postgres"]}, periodSeconds: 5}
        volumeMounts: [{name: data, mountPath: /var/lib/postgresql/data}]
  volumeClaimTemplates:
  - metadata: {name: data}
    spec: {accessModes: [ReadWriteOnce], resources: {requests: {storage: {{ .Values.db.storage }}}}}
EOT
cat > chart/templates/api.yaml <<'EOT'
apiVersion: apps/v1
kind: Deployment
metadata: {name: {{ .Release.Name }}-api}
spec:
  replicas: {{ .Values.api.replicas }}
  selector: {matchLabels: {app: {{ .Release.Name }}-api}}
  template:
    metadata: {labels: {app: {{ .Release.Name }}-api}}
    spec:
      securityContext: {runAsNonRoot: true, seccompProfile: {type: RuntimeDefault}}
      containers:
      - name: api
        image: "{{ .Values.api.image.repository }}:{{ .Values.api.image.tag }}"
        imagePullPolicy: {{ .Values.api.image.pullPolicy }}
        ports: [{containerPort: 8000}]
        env:
        - {name: DB_HOST, value: {{ .Release.Name }}-db}
        - {name: DB_PASSWORD, valueFrom: {secretKeyRef: {name: {{ include "todo.secretName" . }}, key: password}}}
        readinessProbe: {httpGet: {path: /readyz, port: 8000}, periodSeconds: 5}
        livenessProbe: {httpGet: {path: /healthz, port: 8000}, initialDelaySeconds: 10}
        resources: {{- toYaml .Values.api.resources | nindent 10 }}
        securityContext: {allowPrivilegeEscalation: false, capabilities: {drop: [ALL]}}
---
apiVersion: v1
kind: Service
metadata: {name: {{ .Release.Name }}-api}
spec: {selector: {app: {{ .Release.Name }}-api}, ports: [{port: 80, targetPort: 8000}]}
{{- if .Values.autoscaling.enabled }}
---
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata: {name: {{ .Release.Name }}-api}
spec:
  scaleTargetRef: {apiVersion: apps/v1, kind: Deployment, name: {{ .Release.Name }}-api}
  minReplicas: {{ .Values.autoscaling.min }}
  maxReplicas: {{ .Values.autoscaling.max }}
  metrics: [{type: Resource, resource: {name: cpu, target: {type: Utilization, averageUtilization: {{ .Values.autoscaling.cpu }}}}}]
{{- end }}
EOT
cat > chart/templates/web.yaml <<'EOT'
apiVersion: v1
kind: ConfigMap
metadata: {name: {{ .Release.Name }}-web}
data:
  index.html: |
    <!doctype html><title>Todo</title><h1>Todo</h1><ul id=l></ul>

    async function add(){await fetch('/api/todos',{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({title:t.value})});t.value='';load()}
    load()
    </script>
---
apiVersion: apps/v1
kind: Deployment
metadata: {name: {{ .Release.Name }}-web}

  replicas: 2

    metadata: {labels: {app: {{ .Release.Name }}-web}}

      - name: web

---
apiVersion: v1
kind: Service
metadata: {name: {{ .Release.Name }}-web}
spec: {selector: {app: {{ .Release.Name }}-web}, ports: [{port: 80, targetPort: 8080}]}
EOT
cat > chart/templates/ingress.yaml <<'EOT'
{{- if .Values.ingress.enabled }}
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata: {name: {{ .Release.Name }}}
spec:
  ingressClassName: {{ .Values.ingress.className }}
  rules:
  - host: {{ .Values.ingress.host }}

      paths:
      - {path: /api, pathType: Prefix, backend: {service: {name: {{ .Release.Name }}-api, port: {number: 80}}}}

{{- end }}
EOT

apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata: {name: {{ .Release.Name }}-db-only-from-api}

  podSelector: {matchLabels: {app: {{ .Release.Name }}-db}}
  policyTypes: [Ingress]

EOT

6. Deploy, verify, scale, update

bash
kubectl create ns todo
helm install todo chart -n todo --set db.password="$(openssl rand -hex 12)" --wait --timeout 5m
kubectl -n todo get all,ingress,pvc
curl -s -XPOST -H 'Host: todo.lab.local' -H 'content-type: application/json' -d '{"title":"ship it"}' http://127.0.0.1:30080/api/todos
curl -s -H 'Host: todo.lab.local' http://127.0.0.1:30080/api/todos; echo
curl -s -H 'Host: todo.lab.local' http://127.0.0.1:30080/ | head -3

# self-healing and persistence
kubectl -n todo delete pod -l app=todo-api --wait=false
kubectl -n todo delete pod todo-db-0 && kubectl -n todo wait --for=condition=ready pod/todo-db-0 --timeout=120s
curl -s -H 'Host: todo.lab.local' http://127.0.0.1:30080/api/todos; echo   # data still present

# load and autoscale
kubectl -n todo run load --rm -it --image=busybox:1.36 --restart=Never -- sh -c 'for i in $(seq 1 2000); do wget -qO- http://todo-api/api/todos >/dev/null; done'
kubectl -n todo get hpa

# release v1.0.1 with zero downtime
docker build -t todo-api:1.0.1 api/ && docker save todo-api:1.0.1 | ctr -n k8s.io images import -
helm upgrade todo chart -n todo --reuse-values --set api.image.tag=1.0.1 --wait
helm history todo -n todo
helm rollback todo 1 -n todo --wait
