2. Install and verify

bash

3. Learn with no cloud account (local provider)

bash
mkdir -p ~/lab/tf-local && cd ~/lab/tf-local
cat > main.tf <<'EOT'
terraform {
  required_version = ">= 1.6"
  required_providers {
    local  = { source = "hashicorp/local", version = "~> 2.5" }
    random = { source = "hashicorp/random", version = "~> 3.6" }
  }
}
variable "greeting" {
  type    = string
  default = "hello"
}
resource "random_pet" "name" { length = 2 }
resource "local_file" "hello" {
  filename = "${path.module}/${random_pet.name.id}.txt"
  content  = "${var.greeting} from ${random_pet.name.id}\n"
}
output "file" { value = local_file.hello.filename }
EOT
terraform init
terraform fmt && terraform validate
terraform plan -out tfplan
terraform apply tfplan
terraform output; cat *.txt
terraform apply -var greeting=namaste -auto-approve
terraform state list; terraform show | head -20
terraform destroy -auto-approve

4. HCL building blocks

BlockPurposeExample

resourceSomething Terraform managesresource "aws_s3_bucket" "b" {}
dataRead-only lookupdata "aws_ami" "ubuntu" {}
variableInputvariable "env" { type = string }
localsComputed valueslocals { tags = { env = var.env } }

Meta-arguments: count, for_each, depends_on, lifecycle (prevent_destroy, create_before_destroy, ignore_changes). Functions you will use daily: merge, lookup, format, cidrsubnet, templatefile, toset.

5. AWS project: VPC, security group, EC2, S3

Cost
This creates billable resources (EC2, and a NAT gateway if you add one). The example avoids NAT and uses a t3.micro, but still run terraform destroy when finished. Credentials come from aws configure or environment variables (page 10), never from the code.
bash
mkdir -p ~/lab/tf-aws && cd ~/lab/tf-aws
cat > versions.tf <<'EOT'
terraform {
  required_version = ">= 1.6"
  required_providers {
    aws    = { source = "hashicorp/aws", version = "~> 5.0" }
    random = { source = "hashicorp/random", version = "~> 3.6" }
  }
}
provider "aws" {
  region = var.region
  default_tags { tags = { Project = "devops-course", ManagedBy = "terraform" } }
}
EOT
cat > variables.tf <<'EOT'
variable "region" {
  type    = string
  default = "ap-south-1"
}
variable "name" {
  type    = string
  default = "lab"
}
variable "my_ip_cidr" {
  type        = string
  description = "Your public IP, for example 203.0.113.7/32"
}
variable "instance_type" {
  type    = string
  default = "t3.micro"
}
EOT
cat > main.tf <<'EOT'
data "aws_availability_zones" "az" { state = "available" }

data "aws_ami" "ubuntu" {
  most_recent = true
  owners      = ["099720109477"] # Canonical
  filter {
    name   = "name"
    values = ["ubuntu/images/hvm-ssd-gp3/ubuntu-noble-24.04-amd64-server-*"]
  }
}

resource "aws_vpc" "main" {
  cidr_block           = "10.0.0.0/16"
  enable_dns_hostnames = true
  tags                 = { Name = "${var.name}-vpc" }
}
resource "aws_internet_gateway" "igw" { vpc_id = aws_vpc.main.id }
resource "aws_subnet" "public" {
  vpc_id                  = aws_vpc.main.id
  cidr_block              = cidrsubnet(aws_vpc.main.cidr_block, 8, 1)
  availability_zone       = data.aws_availability_zones.az.names[0]
  map_public_ip_on_launch = true
  tags                    = { Name = "${var.name}-public" }
}
resource "aws_route_table" "public" {
  vpc_id = aws_vpc.main.id
  route {
    cidr_block = "0.0.0.0/0"
    gateway_id = aws_internet_gateway.igw.id
  }
}
resource "aws_route_table_association" "public" {
  subnet_id      = aws_subnet.public.id
  route_table_id = aws_route_table.public.id
}

resource "aws_security_group" "web" {
  name   = "${var.name}-web"
  vpc_id = aws_vpc.main.id
  ingress {
    description = "SSH from my IP only"
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = [var.my_ip_cidr]
  }
  ingress {
    description = "HTTP"
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }
  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

resource "aws_instance" "web" {
  ami                    = data.aws_ami.ubuntu.id
  instance_type          = var.instance_type
  subnet_id              = aws_subnet.public.id
  vpc_security_group_ids = [aws_security_group.web.id]
  metadata_options { http_tokens = "required" } # IMDSv2
  root_block_device {
    encrypted   = true
    volume_size = 12
  }
  user_data = <<-EOS
    #!/bin/bash
    apt-get update -y && apt-get install -y nginx
    echo "hello from terraform" > /var/www/html/index.html
  EOS
  tags      = { Name = "${var.name}-web" }
}

resource "random_id" "suffix" { byte_length = 4 }
resource "aws_s3_bucket" "artifacts" { bucket = "${var.name}-artifacts-${random_id.suffix.hex}" }
resource "aws_s3_bucket_public_access_block" "artifacts" {
  bucket                  = aws_s3_bucket.artifacts.id
  block_public_acls       = true
  block_public_policy     = true
  ignore_public_acls      = true
  restrict_public_buckets = true
}
resource "aws_s3_bucket_server_side_encryption_configuration" "artifacts" {
  bucket = aws_s3_bucket.artifacts.id
  rule {
    apply_server_side_encryption_by_default {
      sse_algorithm = "AES256"
    }
  }
}
EOT
cat > outputs.tf <<'EOT'
output "public_ip" { value = aws_instance.web.public_ip }
output "bucket" { value = aws_s3_bucket.artifacts.id }
EOT
MYIP=$(curl -s https://checkip.amazonaws.com)
terraform init
terraform validate
terraform plan -var "my_ip_cidr=${MYIP}/32" -out tfplan
terraform apply tfplan
curl -s http://$(terraform output -raw public_ip)
terraform destroy -var "my_ip_cidr=${MYIP}/32"

6. Remote state and locking

bash
# one-time bootstrap (create the bucket outside the project that uses it)
aws s3api create-bucket --bucket YOUR-UNIQUE-tfstate --region ap-south-1 --create-bucket-configuration LocationConstraint=ap-south-1
aws s3api put-bucket-versioning --bucket YOUR-UNIQUE-tfstate --versioning-configuration Status=Enabled
aws s3api put-public-access-block --bucket YOUR-UNIQUE-tfstate --public-access-block-configuration BlockPublicAcls=true,IgnorePublicAcls=true,BlockPublicPolicy=true,RestrictPublicBuckets=true
cat > backend.tf <<'EOT'
terraform {
  backend "s3" {
    bucket       = "YOUR-UNIQUE-tfstate"
    key          = "lab/terraform.tfstate"
    region       = "ap-south-1"
    encrypt      = true
    use_lockfile = true # native S3 locking in recent Terraform versions; older versions use dynamodb_table
  }
}
EOT
terraform init -migrate-state

7. Modules, workspaces, import

bash
mkdir -p modules/bucket && cat > modules/bucket/main.tf <<'EOT'
variable "name" { type = string }
resource "aws_s3_bucket" "this" { bucket = var.name }
output "arn" { value = aws_s3_bucket.this.arn }
EOT
cat > use-module.tf <<'EOT'
module "logs" {
  source = "./modules/bucket"
  name   = "${var.name}-logs-${random_id.suffix.hex}"
}
EOT
terraform init && terraform plan -var "my_ip_cidr=${MYIP}/32" | tail -5
terraform workspace new staging; terraform workspace list; terraform workspace select default
# adopt an existing resource into state (Terraform 1.5+ import block)
cat > import.tf.example <<'EOT'
import {
  to = aws_s3_bucket.existing
  id = "my-existing-bucket"
}
EOT
terraform plan -generate-config-out=generated.tf   # after renaming the example file to import.tf

8. Quality and safety

bash
terraform fmt -check -recursive
terraform validate
tflint --init && tflint 2>/dev/null || echo "optional: install tflint"
trivy config . 2>/dev/null || echo "optional: install trivy for IaC scanning"
terraform plan -refresh-only      # detect drift without changing anything
terraform state mv aws_instance.web aws_instance.app   # rename without recreating (also edit the .tf)
RuleWhy
Review every plan; save with -outApply exactly what was reviewed
Commit .terraform.lock.hcl, ignore .terraform/ and *.tfstateReproducible providers, no secrets in Git
Pin provider and module versionsAvoid surprise upgrades
Small state per environment and layerSmaller blast radius, faster plans
prevent_destroy on data storesBlock accidental deletion
Checkpoint
You can init, plan, apply and destroy; explain what the state file is; and move state to a remote backend.
