2. Install and prepare a target

bash
pipx install --include-deps ansible 2>/dev/null || pip3 install --user ansible

cat > ansible.cfg <<'EOT'
[defaults]
inventory = inventory.ini
host_key_checking = False
stdout_callback = yaml
[privilege_escalation]

EOT

[web]
localhost ansible_connection=local
[db]
localhost ansible_connection=local
[all:vars]
ansible_python_interpreter=/usr/bin/python3
EOT
ansible-inventory --graph
ansible all -m ping

3. First playbook

bash
cat > site.yml <<'EOT'
- name: Configure web servers
  hosts: web
  vars: {site_title: "Ansible lab", http_port: 8081}
  tasks:
  - name: Install nginx
    ansible.builtin.apt: {name: nginx, state: present, update_cache: true, cache_valid_time: 3600}
  - name: Deploy config
    ansible.builtin.template: {src: site.conf.j2, dest: /etc/nginx/conf.d/lab.conf, mode: "0644"}
    notify: Reload nginx
  - name: Deploy page
    ansible.builtin.copy: {content: "<h1>{{ site_title }}</h1>\n", dest: /var/www/lab/index.html, mode: "0644"}
    when: ansible_facts['os_family'] == "Debian"
  - name: Ensure running
    ansible.builtin.service: {name: nginx, state: started, enabled: true}
  handlers:
  - name: Reload nginx
    ansible.builtin.service: {name: nginx, state: reloaded}
EOT
mkdir -p templates
printf 'server {\n  listen {{ http_port }};\n  root /var/www/lab;\n}\n' > templates/site.conf.j2
mkdir -p /var/www/lab
ansible-playbook site.yml --syntax-check
ansible-playbook site.yml --check --diff      # dry run
ansible-playbook site.yml
ansible-playbook site.yml                       # second run: changed=0 proves idempotency
curl -s localhost:8081

5. Roles (reusable structure)

bash
cd ~/lab/ansible
ansible-galaxy role init roles/webserver
cat > roles/webserver/tasks/main.yml <<'EOT'
- name: Install packages
  ansible.builtin.apt: {name: "{{ web_packages }}", state: present}
- name: Enable service
  ansible.builtin.service: {name: nginx, state: started, enabled: true}
EOT
printf 'web_packages: [nginx, curl]\n' > roles/webserver/defaults/main.yml
printf -- '- hosts: web\n  roles: [webserver]\n' > roles.yml
ansible-playbook roles.yml
ansible-galaxy collection install community.general ansible.posix

7. Hardening playbook example

bash
cat > harden.yml <<'EOT'
- hosts: all
  tasks:
  - ansible.builtin.lineinfile: {path: /etc/ssh/sshd_config.d/99-lab.conf, line: "PermitRootLogin prohibit-password", create: true, mode: "0644"}
    notify: Reload ssh
  - ansible.builtin.apt: {name: [ufw, fail2ban, unattended-upgrades], state: present}
  - community.general.ufw: {rule: allow, name: OpenSSH}
  handlers:
  - {name: Reload ssh, ansible.builtin.service: {name: ssh, state: reloaded}}
EOT
ansible-playbook harden.yml --check
ansible-lint site.yml 2>/dev/null || pip3 install --user ansible-lint
Run --check --diff first
The hardening play touches remote login settings. Test on a disposable machine and keep a second session open.

8. Using Ansible with Kubernetes

bash
ansible-galaxy collection install kubernetes.core
pip3 install --user kubernetes
cat > k8s.yml <<'EOT'
- hosts: localhost
  connection: local
  tasks:
  - kubernetes.core.k8s:
      state: present
      definition:
        apiVersion: v1
        kind: Namespace
        metadata: {name: from-ansible}
EOT
ansible-playbook k8s.yml && kubectl get ns from-ansible && kubectl delete ns from-ansible
Checkpoint
Second playbook run shows changed=0, and you can explain the difference between a role, a handler and a template.
